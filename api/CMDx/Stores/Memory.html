Overview

Process-local key/value store with optional per-key TTL. Thread-safe via an internal mutex. Suitable for tests and single-process dev; not shared across workers.

def decr(key, ttl: nil)
  key = Stores.key!(key)
  Stores.ttl!(ttl) if ttl

  @mutex.synchronize do
    entry = @entries[key]
    if entry && expired?(entry)
      @entries.delete(key)
      entry = nil
    end

    count = entry ? entry[:value].to_i : 0
    count = count.positive? ? count - 1 : 0
    expires_at = entry&.fetch(:expires_at, nil)

    if count.zero?
      @entries.delete(key)
    else
      @entries[key] = { value: count.to_s, expires_at: }
    end

    count
  end
end

def del_if(key, expected)
  key = Stores.key!(key)
  expected = expected.to_s
  @mutex.synchronize do
    entry = @entries[key]
    return false unless entry && !expired?(entry)
    return false unless entry[:value] == expected

    @entries.delete(key)
    true
  end
end

def get(key)
  key = Stores.key!(key)
  @mutex.synchronize do
    entry = @entries[key]
    return unless entry
    return if expired?(entry)

    entry[:value]
  end
end

def incr(key, ttl: nil)
  key = Stores.key!(key)
  ttl = Stores.ttl!(ttl) if ttl

  @mutex.synchronize do
    entry = @entries[key]
    if entry && expired?(entry)
      @entries.delete(key)
      entry = nil
    end

    count = entry ? entry[:value].to_i + 1 : 1
    expires_at = (Util.monotonic_now + ttl if ttl)

    @entries[key] = { value: count.to_s, expires_at: }
    count
  end
end

def set(key, value, ttl: nil, nx: false, xx: false)
  key = Stores.key!(key)
  ttl = Stores.ttl!(ttl) if ttl
  Stores.nx_xx!(nx, xx)

  expires_at = ttl ? Util.monotonic_now + ttl : nil
  @mutex.synchronize do
    entry = @entries[key]
    present = entry && !expired?(entry)

    return false if nx && present
    return false if xx && !present

    @entries[key] = { value: value.to_s, expires_at: }
    true
  end
end

def ttl(key)
  key = Stores.key!(key)
  @mutex.synchronize do
    entry = @entries[key]
    return unless entry
    return if expired?(entry)

    expires_at = entry[:expires_at]
    return unless expires_at

    remaining = expires_at - Util.monotonic_now
    remaining.positive? ? remaining.ceil : nil
  end
end
