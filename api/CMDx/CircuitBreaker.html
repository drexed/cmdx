Overview

Declared via Task.circuit_breaker. After inputs resolve, consults shared store counters before the retry_on loop; opens after repeated failures and short-circuits until cool_off elapses.

Store keys come from StoreKey#task_key with class_scoped: true, plus :failures and :open suffix segments (see StoreKey#compose).

def build(new_key, new_options)
  return self if new_key.nil? && new_options.empty?

  merged_key =
    if new_key.nil?
      enabled? || new_options.empty? ? @key : StoreKey::CLASS_KEY
    else
      new_key
    end
  merged_options = @options.merge(new_options)

  self.class.new(merged_key, merged_options)
end

def process(task)
  return yield unless enabled?
  return yield unless Util.satisfied?(condition_if, condition_unless, task)

  store_key = StoreKey.task_key(task, namespace:, resolver: key, class_scoped: true)
  return yield if store_key.nil?

  open_key     = StoreKey.compose(store_key, OPEN)
  failures_key = StoreKey.compose(store_key, FAILURES)
  open!(task, open_key) if circuit_open?(open_key)

  begin
    yield
  rescue StandardError => e
    record_failure(e, failures_key, open_key, task)
    raise e
  else
    clear_failures(failures_key, task)
  end
end
